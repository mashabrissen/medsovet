function saveToHistory(question, answer) {
    chatHistory.unshift({
        q: question,
        a: answer.substring(0, 100) + '...',
        fullA: answer,
        time: new Date().toLocaleString('ru-RU')
    });
    if (chatHistory.length > 50) chatHistory.pop();
    localStorage.setItem('chatHistory', JSON.stringify(chatHistory));
}

function renderHistory() {
    const list = document.getElementById('historyList');
    if (chatHistory.length === 0) {
        list.innerHTML = '<p style="color: #999;">История пуста</p>';
        return;
    }
    list.innerHTML = chatHistory.map((item, i) => `
        <div class="history-item" onclick="loadFromHistory(${i})">
            <div class="q">${item.q}</div>
            <div class="a">${item.a}</div>
        </div>
    `).join('');
}

function loadFromHistory(index) {
    const item = chatHistory[index];
    addMessage(item.q, true);
    setTimeout(() => addMessage(item.fullA, false), 300);
    document.getElementById('historyModal').style.display = 'none';
}

function findAnswer(query) {
    const q = query.toLowerCase();
    let bestMatch = null;
    let bestScore = 0;

    for (const [key, data] of Object.entries(medicalKnowledge)) {
        let score = 0;
        
        // Проверка ключевых слов
        for (const keyword of data.keywords) {
            if (q.includes(keyword)) {
                score += keyword.length; // Длинные совпадения ценнее
            }
        }
        
        // Частичное совпадение с ключом
        if (q.includes(key)) score += 2;
        
        if (score > bestScore) {
            bestScore = score;
            bestMatch = data;
        }
    }

    // Порог: нужно хотя бы одно совпадение
    if (bestScore > 1) {
        return bestMatch.response;
    }

    return null;
}

// ============================================
// ГЕНЕРАЦИЯ ОТВЕТА (ЛОКАЛЬНАЯ)
// ============================================
function generateResponse(userMessage) {
    const localAnswer = findAnswer(userMessage);
    
    if (localAnswer) {
        return localAnswer;
    }

    // Умный fallback на основе ключевых слов
    const q = userMessage.toLowerCase();
    
    if (q.includes('почему') || q.includes('причина')) {
        return `Интересный вопрос. Для точного ответа мне нужно больше деталей.\n\nОпишите подробнее:\n• Где именно проблема?\n• Когда началось?\n• Что провоцирует/облегчает?\n\nА пока — универсальный совет: при любых непонятных симптомах начните с визита к терапевту.`;
    }
    
    if (q.includes('что делать') || q.includes('помогите')) {
        return `Чтобы дать точный совет, уточните детали.\n\n**Общий алгоритм:**\n1. Оцените серьёзность симптома\n2. При тревожных признаках — к врачу или скорая\n3. Не занимайтесь самолечением при непонятных симптомах\n\nОпишите ситуацию подробнее, и я смогу помочь конкретнее.`;
    }
    
    if (q.length < 10) {
        return `Опишите, пожалуйста, подробнее. Что именно беспокоит? Какие симптомы?`;
    }

    return `Спасибо за вопрос. Я проанализировал, но для точного ответа рекомендую:\n\n• Обратиться к терапевту для очного осмотра\n• При болях — не принимать сильнодействующие препараты без назначения\n• Вести дневник симптомов до визита к врачу\n\nПопробуйте переформулировать вопрос — возможно, я смогу дать более конкретный ответ на основе своей базы знаний.`;
}

function addMessage(text, isUser = false) {
    const messageDiv = document.createElement('div');
    messageDiv.className = `message ${isUser ? 'user-message' : 'bot-message'}`;
    messageDiv.innerHTML = `
        <div class="message-avatar">${isUser ? '👤' : '🤖'}</div>
        <div class="message-content">${formatMessage(text)}</div>
    `;
    chatMessages.appendChild(messageDiv);
    chatMessages.scrollTop = chatMessages.scrollHeight;
}

function formatMessage(text) {
    return text
        .replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>')
        .replace(/• (.*?)(?=\n|$)/g, '<li>$1</li>')
        .replace(/(<li>.*<\/li>)/s, '<ul>$1</ul>')
        .replace(/\n/g, '<br>')
        .replace(/🚩/g, '<span style="color:var(--danger);">🚩</span>')
        .replace(/🚑/g, '<span style="color:var(--danger);">🚑</span>');
}

// ============================================
async function sendMessage() {
    const text = userInput.value.trim();
    if (!text) return;

    sendBtn.disabled = true;
    userInput.disabled = true;

    addMessage(text, true);
    const question = text;
    userInput.value = '';
    charCount.textContent = '0';
    userInput.style.height = 'auto';

    showTyping();

    // Имитация задержки для естественности
    await new Promise(resolve => setTimeout(resolve, 600 + Math.random() * 800));

    const response = generateResponse(question);
    
    hideTyping();
    addMessage(response, false);
    
    // Сохраняем в историю
    saveToHistory(question, response);

    sendBtn.disabled = false;
    userInput.disabled = false;
    userInput.focus();
}
